WHILEFORM APP CODE:
namespace WhileFormApp
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnStart_Click(object sender, EventArgs e)
        {
            while (picHero.Top < picCat.Top - 50)
            {
                picHero.Top += 10;
                Application.DoEvents();
                System.Threading.Thread.Sleep(100);
            }
            lblStatus.Text = "You have reached the cat!";
        }    
    }
}

namespace TreasureHuntApp
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnPlay_Click(object sender, EventArgs e)
        {
            int treasure = 1;

            lstScore.Items.Clear();

            while (treasure <= 10)
            {
                lstScore.Items.Add("Treasure " + treasure + " collected!");
                treasure++;
            }
        }
    }
}

namespace CoinCollector
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnCollect_Click(object sender, EventArgs e)
        {
            int coins = 1;
            lblCoins.Text = "";

            do
            {
                lblCoins.Text += "Collected coin #" + coins + "\n";
                coins++;
            }
            while (coins <= 5);
        }
    }
}

MONSTERBATTLE CODE:

namespace MonsterBattle
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnFight_Click(object sender, EventArgs e)
        {
            int monsterHP = 20;
            lblResult.Text = "";

            do
            {
                monsterHP -= 5;
                lblResult.Text += "Monster HP left: " + monsterHP + "\n";
            }
            while (monsterHP > 0);

            lblResult.Text += "Monster defeated!";
        }
    }
}

CHOOSEAWEAPON CODE: 

namespace ChooseWeapon
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnAttack_Click(object sender, EventArgs e)
        {
            string selectedWeapon = cmbWeapon.Text;

            switch (selectedWeapon)
            {
                case "Sword":
                    lblResult.Text = "You slashed with your Sword!";
                    break;
                case "Bow":
                    lblResult.Text = "You shot an arrow with your Bow!";
                    break;
                case "Magic":
                    lblResult.Text = "You cast a fire spell!";
                    break;
                default:
                    lblResult.Text = "Please select a weapon!";
                    break;
            }
        }
    }
}

{

    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnOpen_Click(object sender, EventArgs e)
        {
            string doorChoice = txtDoor.Text;

            switch (doorChoice)
            {
                case "1":
                    lblResult.Text = "You found a chest full of gold!";
                    break;
                case "2":
                    lblResult.Text = "A sleeping dragon is inside!";
                    break;
                case "3":
                    lblResult.Text = "You found a secret exit!";
                    break;
                default:
                    lblResult.Text = "Invalid door! Choose 1, 2, or 3.";
                    break;
            }
        }
    }
}

HEALTHCOLLECTOR CODE: 

namespace HealthCollector
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnHeal_Click(object sender, EventArgs e)
        {
            progressBar1.Value = 0;

            for (int health = 0; health <= 100; health += 20)
            {
                progressBar1.Value = health;
                lblHealth.Text = "❤️ Health: " + health + "%";

                Application.DoEvents();
                System.Threading.Thread.Sleep(500);
            }

            MessageBox.Show("Full Health!");
        }
    }
}
